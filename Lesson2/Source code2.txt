class Student {
  constructor(name, scores) {
    this.name = name;
    this.scores = scores;
  }

  getAverage() {
    let sum = 0;
    for (let i = 0; i < this.scores.length; i++) {
      sum += this.scores[i];
    }
    return sum / this.scores.length;
  }

  getLetterGrade() {
    const avg = this.getAverage();
    if (avg >= 90) {
      return "A";
    } else if (avg >= 80) {
      return "B";
    } else if (avg >= 70) {
      return "C";
    } else if (avg >= 60) {
      return "D";
    } else {
      return "F";
    }
  }
}

function analyzeClass(students) {
  console.log("------- STUDENT REPORT -------");

  let totalClassScore = 0;
  let topStudent = null;
  let highestAverage = -1;

  for (const student of students) {
    const avg = student.getAverage();
    const grade = student.getLetterGrade();

    totalClassScore += avg;

    if (avg > highestAverage) {
      highestAverage = avg;
      topStudent = student;
    }

    console.log(`${student.name} - Average: ${avg.toFixed(1)} | Grade: ${grade}`);
  }

  const classAverage = totalClassScore / students.length;

  console.log("------------------------------");
  console.log(`Class Average: ${classAverage.toFixed(1)}`);
  if (topStudent) {
    console.log(`Top Student: ${topStudent.name} with ${highestAverage.toFixed(1)}`);
  }
}

const studentData = [
  { name: "Lindsay", scores: [95, 93, 98] },
  { name: "Alexa", scores: [89, 85, 83] },
  { name: "Chester", scores: [72, 77, 75] },
  { name: "Jenica", scores: [62, 58, 57] }
];

const students = studentData.map((data) => new Student(data.name, data.scores));
analyzeClass(students);
