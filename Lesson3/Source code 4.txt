// ================================================================
// PART A: STACK
// ================================================================

class Stack {
  // Initialize an empty array for internal storage
  constructor() {
    this.items = [];
  }

  // Add an element to the top of the stack
  push(element) {
    this.items.push(element);
  }

  // Remove and return the top element
  pop() {
    if (this.isEmpty()) return null;
    return this.items.pop();
  }

  // Return the top element without removing it
  peek() {
    if (this.isEmpty()) return null;
    return this.items[this.items.length - 1];
  }

  // Return true if the stack contains no elements
  isEmpty() {
    return this.items.length === 0;
  }

  // Return the total number of elements in the stack
  size() {
    return this.items.length;
  }

  // Return all elements as a comma-separated string
  toString() {
    return this.items.join(', ');
  }
}

console.log('\nPART A: UNDO HISTORY');

const actionHistory = new Stack();

// 1. Push the three required actions
actionHistory.push('Opened Ticket #101');
actionHistory.push('Assigned Ticket #101');
actionHistory.push('Closed Ticket #101');

// 2. Display peek, pop, remaining actions, and size
console.log(`Top Action: ${actionHistory.peek()}`);
console.log(`Undo Action: ${actionHistory.pop()}`);
console.log(`Remaining Actions: ${actionHistory.toString()}`);
console.log(`Stack Size: ${actionHistory.size()}`);

class Queue {
  // Initialize an empty array for internal storage
  constructor() {
    this.items = [];
  }

  // Add an element at the back of the queue
  enqueue(element) {
    this.items.push(element);
  }

  // Remove and return the front element
  dequeue() {
    if (this.isEmpty()) return null;
    return this.items.shift();
  }

  // Return the front element without removing it
  front() {
    if (this.isEmpty()) return null;
    return this.items[0];
  }

  // Return true if the queue contains no elements
  isEmpty() {
    return this.items.length === 0;
  }

  // Return the total number of elements in the queue
  size() {
    return this.items.length;
  }

  // Return all elements as a comma-separated string
  toString() {
    return this.items.join(', ');
  }
}

class PriorityQueue {
  // Initialize an empty array for internal storage
  constructor() {
    this.items = [];
  }

  // Insert item based on priority while preserving arrival order for equal priorities
  enqueue(request, priority) {
    const queueElement = { request, priority };
    let added = false;

    // Strict inequality (<) ensures equal priorities retain FIFO arrival order
    for (let i = 0; i < this.items.length; i++) {
      if (queueElement.priority < this.items[i].priority) {
        this.items.splice(i, 0, queueElement);
        added = true;
        break;
      }
    }

    // Append to end if priority is lower or equal to all existing items
    if (!added) {
      this.items.push(queueElement);
    }
  }

  // Remove and return the first (highest priority) item
  dequeue() {
    if (this.size() === 0) return null;
    return this.items.shift();
  }

  // Return total count of items
  size() {
    return this.items.length;
  }

  // Display every request and priority
  print() {
    this.items.forEach((item) => {
      console.log(`${item.request}, Priority ${item.priority}`);
    });
  }
}

class Node {
  // Initialize node with value and empty next pointer
  constructor(element) {
    this.element = element;
    this.next = null;
  }
}

class LinkedList {
  // Initialize list tracking size and head pointer
  constructor() {
    this.length = 0;
    this.head = null;
  }

  // Append node at the end of the list
  append(element) {
    const node = new Node(element);
    let current;

    if (this.head === null) {
      this.head = node;
    } else {
      current = this.head;
      while (current.next) {
        current = current.next;
      }
      current.next = node;
    }
    this.length++;
  }

  // Insert node at specific index position
  insert(position, element) {
    if (position < 0 || position > this.length) return false;

    const node = new Node(element);
    let current = this.head;
    let previous;
    let index = 0;

    if (position === 0) {
      node.next = current;
      this.head = node;
    } else {
      while (index++ < position) {
        previous = current;
        current = current.next;
      }
      node.next = current;
      previous.next = node;
    }
    this.length++;
    return true;
  }

  // Remove node at specific index position
  removeAt(position) {
    if (position < 0 || position >= this.length) return null;

    let current = this.head;
    let previous;
    let index = 0;

    if (position === 0) {
      this.head = current.next;
    } else {
      while (index++ < position) {
        previous = current;
        current = current.next;
      }
      previous.next = current.next;
    }
    this.length--;
    return current.element;
  }

  // Traverse list and return matching index or -1 if absent
  indexOf(element) {
    let current = this.head;
    let index = 0;

    while (current) {
      if (current.element === element) return index;
      index++;
      current = current.next;
    }
    return -1;
  }

  // Remove node matching target element value
  remove(element) {
    const index = this.indexOf(element);
    return this.removeAt(index);
  }

  // Return true when no nodes exist
  isEmpty() {
    return this.length === 0;
  }

  // Return count of nodes
  size() {
    return this.length;
  }

  // Traverse and return all elements separated by newlines
  toString() {
    let current = this.head;
    let string = '';

    while (current) {
      string += current.element + (current.next ? '\n' : '');
      current = current.next;
    }
    return string;
  }
}
